            pos.y += vel.y;
            visualX = pos.x;
            
            vel.x *= (1 - profile.fric);
            vel.y *= (1 - profile.fric);

            if (Math.abs(vel.x) < 0.05 && Math.abs(vel.y) < 0.05) {
                animationActive = false;
            }
        }

        let currentScale = 1 + (pos.z * 0.04);
        ball.setAttribute('cx', visualX);
        ball.setAttribute('cy', pos.y);
        ball.setAttribute('r', baseRadius * currentScale);
        
        let shadowX = visualX + (pos.z * 0.5);
        let shadowY = pos.y + (pos.z * 0.8);
        shadow.setAttribute('cx', shadowX);
        shadow.setAttribute('cy', shadowY);
        shadow.setAttribute('fill', profile.vz === 0 ? 'none' : 'rgba(0,0,0,0.6)');
        
        tracerPath += ` L ${visualX} ${pos.y}`;
        tracer.setAttribute('d', tracerPath);

        if (animationActive) {
            requestAnimationFrame(physicsLoop);
        } else {
            setTimeout(() => {
                tracer.setAttribute('d', "");
                currentShotIndex = (currentShotIndex + 1) % shotProfiles.length;
                initShot();
            }, 2500);
        }
    }

    initShot();
